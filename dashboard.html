Upload a custom Python rule for complex conditions
Upload a BaseRule subclass (.py)

Use this for custom logic that the standard condition types can't express — such as relation expansion, duration calculation, multi-field aggregation, or cross-collection checks. The class must extend BaseRule and implement a detect() method.

Working template
Copy this, fill in your logic, save as my_rule.py, then upload.
from app.rule_definitions.base_rule import BaseRule

class MyCustomRule(BaseRule):
    name        = "my_custom_rule"
    description = "Describe what this rule detects."

    # These params appear as form fields when creating the rule.
    # datasource_type / url / admin_email / admin_password are required
    # so the scheduler can connect to your data source.
    params_schema = [
        {"key": "datasource_type", "label": "Datasource Type", "type": "text",
         "default": "pocketbase", "hint": "pocketbase | sqlserver | mongodb"},
        {"key": "url",             "label": "PocketBase URL",  "type": "text",
         "default": "http://localhost:8090", "hint": "Base URL of your PocketBase"},
        {"key": "admin_email",     "label": "Admin Email",     "type": "text",
         "default": "", "hint": "PocketBase admin email"},
        {"key": "admin_password",  "label": "Admin Password",  "type": "text",
         "default": "", "hint": "PocketBase admin password"},
        {"key": "collection",      "label": "Collection Name", "type": "text",
         "default": "my_collection", "hint": "Collection to monitor"},
    ]

    def __init__(self, datasource, notifiers, collection="my_collection"):
        super().__init__(notifiers)
        self.datasource = datasource
        self.collection = collection

    def detect(self):
        records = self.datasource.fetch({
            "collection": self.collection,
            "sort": "-created",
            "per_page": 50,
        })
        if not records:
            return []

        # Example: alert on every record unconditionally
        return [
            {
                "message": f"New record in {self.collection}: {r.get('id')}",
                "data": r,
            }
            for r in records
        ]
